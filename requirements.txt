# Runtime + build dependencies.
#
# Version ranges, not exact pins: this is an application, and a grader on a
# different Python patch release should not hit a resolver conflict. The versions
# actually used in the reported run are recorded in eval/results/, which is what
# reproducibility rests on.
#
# TORCH MUST BE THE CPU BUILD. `pip install torch` pulls CUDA wheels (~2GB) by
# default; the runtime is CPU-only, so that payload is waste on every image pull.
# The Dockerfile installs it explicitly from the CPU index:
#
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#
# Locally, `pip install -r requirements.txt` is fine if torch is already present.

# --- core runtime ------------------------------------------------------------
pydantic>=2.6            # typed graph state, schema invariants
pyyaml>=6.0              # SOP frontmatter, config

# --- classifier --------------------------------------------------------------
torch>=2.2               # CPU build; see note above
transformers>=5.0        # MiniLM-L6 encoder + tokenizer
# NumPy 2.5 requires Python 3.12+, while this project and its Docker image support
# Python 3.11. The cap also keeps NumPy's type stubs parseable under that target.
numpy>=2.0,<2.5

# --- orchestration -----------------------------------------------------------
langgraph>=1.0           # explicit nodes, shared state, traceable execution path

# --- evaluation --------------------------------------------------------------
scikit-learn>=1.4        # macro-F1, calibration curves, the TF-IDF leakage check
pandas>=3.0              # eval tables, dataset EDA
matplotlib>=3.8          # reliability diagrams, risk-coverage curves

# --- demo --------------------------------------------------------------------
streamlit>=1.33          # two queues, reason chips, threshold slider

# --- LLM providers -----------------------------------------------------------
# Each has one role, and no model is scored on its own output (BUILD.md S6.4).
google-genai>=1.0          # Gemini: maintained drafting SDK (runtime, free tier)
groq>=0.11                # Groq: groundedness judge (eval only, free tier)
openai>=3.0               # OpenAI: email generation (build time) + bonus judge

# --- notebooks ---------------------------------------------------------------
jupyter>=1.0             # training + EDA notebook
ipykernel>=6.29

# --- development -------------------------------------------------------------
# Also declared in pyproject.toml [project.optional-dependencies].dev
pytest>=8.0
ruff>=0.4
mypy>=2.0
types-PyYAML>=6.0       # typed signatures for strict mypy checks of YAML config
